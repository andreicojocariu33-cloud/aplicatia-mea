function syncModalRotations() { const dialogs = document.querySelectorAll(".fs-modal-dialog"); dialogs.forEach((dlg) => { dlg.classList.remove("fs-rot-ccw", "fs-rot-cw"); if (isFocusMode) { dlg.classList.add(rotationMode === "cw" ? "fs-rot-cw" : "fs-rot-ccw"); } }); } function applyRotationClass() { const shell = document.getElementById("fullscreen-overlay-shell"); if (!shell) return; shell.classList.remove("rotated-cw", "rotated-ccw"); if (isFocusMode) { shell.classList.add(rotationMode === "cw" ? "rotated-cw" : "rotated-ccw"); } syncModalRotations(); } function enterFullscreenSchedule() { if (isFocusMode) return; isFocusMode = true; if (viewMode !== "table") viewMode = "table"; const shell = document.getElementById("fullscreen-overlay-shell"); const fsTopBar = document.getElementById("fullscreen-top-bar"); const tableContainer = document.getElementById("table-view-container"); const scrollWrapper = document.getElementById("table-scroll-wrapper"); document.body.classList.add("fullscreen-schedule-active"); if (shell) { shell.classList.add("active"); applyRotationClass(); } if (fsTopBar) fsTopBar.classList.remove("hidden"); if (tableContainer) { tableContainer.classList.remove("rounded-2xl", "border"); tableContainer.classList.add("rounded-none", "border-0"); } if (scrollWrapper) { scrollWrapper.classList.remove("max-h-[78vh]"); scrollWrapper.style.height = "100%"; scrollWrapper.style.maxHeight = "none"; } renderAll(); } function exitFullscreenSchedule() { if (!isFocusMode) return; isFocusMode = false; try { if (document.fullscreenElement && document.exitFullscreen) { document.exitFullscreen().catch(() => {}); } } catch (e) {} const shell = document.getElementById("fullscreen-overlay-shell"); const fsTopBar = document.getElementById("fullscreen-top-bar"); const tableContainer = document.getElementById("table-view-container"); const scrollWrapper = document.getElementById("table-scroll-wrapper"); document.body.classList.remove("fullscreen-schedule-active"); if (shell) shell.classList.remove("active", "rotated-cw", "rotated-ccw"); syncModalRotations(); if (fsTopBar) fsTopBar.classList.add("hidden"); if (tableContainer) { tableContainer.classList.add("rounded-2xl", "border"); tableContainer.classList.remove("rounded-none", "border-0"); } if (scrollWrapper) { scrollWrapper.style.height = ""; scrollWrapper.style.maxHeight = ""; scrollWrapper.classList.add("max-h-[78vh]"); } renderAll(); }
